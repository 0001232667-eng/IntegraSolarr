Código C++ para ESP32

Programe seu ESP32 para mandar dados automaticamente:

#include <WiFi.h>
#include <HTTPClient.h>

void setup() {
  WiFi.begin("SUA_REDE_WIFI", "SUA_SENHA");
}

void loop() {
  if(WiFi.status() == WL_CONNECTED){
    HTTPClient http;
    http.begin("https://sua-api.com/api/leituras");
    http.addHeader("Content-Type", "application/json");

    String json = "{\"potencia\":1.85, \"carga\":78, \"tensao\":220}";
    int response = http.POST(json);
    http.end();
  }
  delay(10000); // Envia a cada 10s
}
